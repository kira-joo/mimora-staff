# mimora-staff — environment (Mimora backend + staff console)
#
# Install:  cp docs/env-example.txt .env.example      then copy .env.example to .env and fill it in.
# Placeholders only — never put a real value in this file or in .env.example. Locally the values
# live in .env (git-ignored); in a deployment they live in the host's secret store
# (docs/deployment.md). Every variable is SERVER-ONLY: none is NEXT_PUBLIC_, none reaches a browser.
#
# Validated at boot by src/server/core/config/env-validation.ts (the single declaration of every
# variable; a test fails if the code reads one that table does not declare):
#   REQUIRED    a production server refuses to start without it, naming the variable (never its value)
#   CAPABILITY  switches a feature on. Unset = the feature is OFF, announced loudly once at startup.
#               Leave it EMPTY (do not delete the line) until you configure it. A half-configured
#               group (some of its variables set, some not) is reported by name.
#   OPTIONAL    has a safe default
# Tags
#   [secret]          generate it (openssl rand -base64 48); never reuse one value for two variables
#   [pair]            must equal the same-named value in mimora-client's environment
#   [per environment] differs between local, preview and production

# ── Database — REQUIRED ───────────────────────────────────────────────
# [per environment] [secret in production] Must reach a MongoDB REPLICA SET: order placement is one
# transaction. Locally `docker compose up -d` provides rs0 on port 27020. In production use a
# managed replica set, e.g. a MongoDB Atlas mongodb+srv:// string (it carries credentials).
MONGODB_URI=mongodb://localhost:27020/?replicaSet=rs0&directConnection=true
# [per environment] Database name. Two environments must never share one.
# Local Mimora data lives in `store-platform-staff` (an inherited name, adopted as STORE_ID mimora;
# renaming it is a planned migration, not an edit here).
MONGODB_DB=store-platform-staff

# ── Authentication — REQUIRED ─────────────────────────────────────────
# [secret] [per environment] Two audiences, two secrets: each at least 32 characters, different from
# each other and from every other secret here. Rotating one signs that audience out everywhere.
STAFF_JWT_SECRET=
CUSTOMER_JWT_SECRET=
# OPTIONAL. Customer access-token lifetime: a number plus s/m/h/d. Default 15m.
CUSTOMER_ACCESS_TOKEN_TTL=

# ── Store identity — REQUIRED ─────────────────────────────────────────
# [pair: mimora-client STORE_ID] Must equal the marker in the database (written once by
# scripts/admin/adopt-store-identity.mjs). The backend refuses any database whose marker names
# another store, or has none. Never change it.
STORE_ID=mimora
# [pair: mimora-client STORE_TIMEZONE] IANA time zone. Cutoffs, opening hours, "today" and reports use it.
STORE_TIMEZONE=Africa/Cairo
# CAPABILITY. Country calling code for national phone numbers, digits only. Egypt is 20.
# Unset: numbers are stored as typed. Read scripts/migrations/README.md before changing it once
# customers exist.
STORE_PHONE_COUNTRY_CODE=20

# ── Storefront cache invalidation — CAPABILITY, both or neither ───────
# Unset: the storefront's prerendered catalog pages lag Staff edits by up to 300 seconds.
# The storefront's receiving URL: https://<mimora-client domain>/api/revalidate
STOREFRONT_REVALIDATE_URL=
# [secret] [pair: mimora-client REVALIDATE_SECRET]
REVALIDATE_SECRET=

# ── Image and receipt uploads, Cloudinary — CAPABILITY, all three or none ──
# Unset: catalog/brand image uploads and payment-receipt uploads (InstaPay, wallet and bank
# transfer orders) answer 503 with a clear message. Public images go under `mimora/store/`;
# receipts are private assets fetched only through short-lived signed URLs. Use a Cloudinary
# product environment dedicated to Mimora.
# Cloudinary Console → Settings (gear) → API Keys / Product environment:
#   cloud name = "Cloud name"; API key and secret = a key row under "API Keys" (Generate New API Key).
CLOUDINARY_CLOUD_NAME=
CLOUDINARY_API_KEY=
# [secret]
CLOUDINARY_API_SECRET=

# ── Email, Nodemailer over SMTP — CAPABILITY ──────────────────────────
# Account emails (verify, reset, staff invite), the order emails enabled in Settings →
# Notifications, and newsletter confirmations. Setting ANY of these asks for the rest. Unset:
# nothing is sent; each would-be message is logged by template name only.
SMTP_HOST=
# 587 (STARTTLS) or 465 (implicit TLS).
SMTP_PORT=
# true only with 465; empty follows the port.
SMTP_SECURE=
# User and password together, or neither (unauthenticated relay).
SMTP_USER=
# [secret]
SMTP_PASS=
# Sender, e.g. Mimora <orders@your-domain>. The domain must be verified with the SMTP provider (SPF/DKIM).
MAIL_FROM=
# OPTIONAL within this group: where customer replies go.
MAIL_REPLY_TO=
# Public origins used to build links in emails. Required once SMTP is configured.
# STOREFRONT_URL = mimora-client's public origin; BACKOFFICE_URL = this console's public origin.
STOREFRONT_URL=
BACKOFFICE_URL=

# ── Scheduled jobs — CAPABILITY ───────────────────────────────────────
# The unpaid-order sweep (every 15 min: cancels proof-required orders past their window, releases
# stock) and the orphan-image sweep (daily). Both secrets unset: job routes answer 404; jobs are OFF.

# [secret] Any other scheduler: JOBS_SECRET, sent as the x-jobs-secret header (or Bearer).
